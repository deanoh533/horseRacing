## 🧮 알고리즘 (확정)

```javascript
function calculateJockeyFormScore(jkNo, todayDate) {
  // 최근 30일 기수의 모든 경주 (서울 + 부산경남 통합)
  const thirtyDaysAgo = subtractDays(todayDate, 30);
  const recent = db.query(`
    SELECT ord FROM race_entries
    WHERE jcky_no = $1 AND race_date >= $2 AND ord IS NOT NULL
  `, [jkNo, thirtyDaysAgo]);
  
  if (recent.length < 5) return 0.5; // 데이터 부족 (5회 미만)
  
  const top1 = recent.filter(r => r.ord === 1).length;
  const top3 = recent.filter(r => r.ord <= 3).length;
  const total = recent.length;
  
  // 메인: 입상 비율 (사용자 노하우 ⭐)
  const top3Rate = top3 / total;
  
  // 보너스: 1등 가중 (1등이 3등보다 약간 더 가치)
  const top1Bonus = (top1 / total) * 0.2;
  
  return Math.min(1.0, top3Rate + top1Bonus);
}
```
